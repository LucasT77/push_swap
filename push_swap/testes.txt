#include <stdio.h>
#include <stdlib.h>

void print_array(int *array, int size)
{
    int i;

    i = 0;
    while (i < size)
        printf("%d\n", array[i++]);
}

int	*rotate(int	*stack, int size)
{
	int	i;

	i = size - 1;
	while (i > 0)
	{
		stack[0] ^= stack[i];		
		stack[i] ^= stack[0];
		stack[0] ^= stack[i];
		i--;
	}
	return (stack);
}

int main()
{
    int size = 5;
    int *x;
    int i;

    x = malloc(sizeof(int) * size);
    i = 0;
    while (i < size)
    {
        x[i] = i + 1;
        i++;
    }
    print_array(x, size);
    x = rotate(x, size);
    printf("\n\n");
    print_array(x, size);
    free (x);
    return 0;
}

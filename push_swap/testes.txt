int	*rotate(int	*stack, int size)
{
	int	i;

	i = size - 1;
	while (i > 0)
	{
		stack[0] ^= stack[i];		
		stack[i] ^= stack[0];
		stack[0] ^= stack[i];
		i--;
	}
	return (stack);
}

void	push(int **taken, int **given, int size_taken, int size_given)
{
	int	*taken_tmp;
	int	*given_tmp;
	int	i;

	taken_tmp = (*taken);
	given_tmp = (*given);
	(*taken) = malloc(sizeof(int) * (size_taken - 1));
	(*given) = malloc(sizeof(int) * (size_given + 1));
	(*given)[0] = taken_tmp[0];
	i = 1;
	while (i <= size_given)
	{
		(*given)[i] = given_tmp[(i-1)];
		i++;
	}
	i = 0;
	while (i < size_taken - 1)
	{
		(*taken)[i] = taken_tmp[(i+1)];
		i++;
	}
	free(taken_tmp);
	free(given_tmp);
}

#include <stdio.h>

int	ft_power(int nb, int power)
{
	int	result;

	if (power < 0)
		return (0);
	result = 1;
	if (power == 0)
		return (1);
	while (power >= 1)
	{
		result *= nb;
		power--;
	}
	return (result);
}

int main()
{
    int x;
    int bit;
    int cont;
    
    cont = 0;
    bit = 0;
    x = 7;
    printf("x = %d\n", x);
    while (bit < 32)
    {
        cont += (x >> bit & 1) * ft_power(2, bit);
        if (cont == x)
            break ;
        bit++;
    }
    printf ("bit = %d\n", bit);
    printf("cont = %d\n", cont);

    return 0;
}

void    to_binary(int num)
{
    int *arr;
    int i;

    if (num == 0)
	{
		ft_printf("%d", num);
		return ;
	}
	arr = malloc(sizeof(int));
    i = 0;
    while (num > 0)
    {
        if (num % 2 > 0)
            arr[i] = 1;
        else
            arr[i] = 0;
        i++;
        num /= 2;
    }
    while (i > 0)
    {
        ft_printf("%d", arr[i - 1]);
        i--;
    }
}

{
	int	i;

	i = 0;
	while (i < size)
	{
		to_binary(array[i]);
		ft_printf("\n");
		i++;
	}
}
